# Chapter 97: Reference Implementation — Text Resume Generator

Converts a simple markdown-ish resume into a clean, printable single-page HTML
resume. Recognizes `# Name`, `## Section`, `- item`, and `**bold**`. Demonstrates
string parsing, HTML/CSS generation, and `@media print` styling.

## 97.1 The program (`resume.z`)

```zen
import sys
import pathlib
import os

func read_all(path) {
    with open(path, "r") as f { return f.read() }
}

func write_all(path, text) {
    let p = pathlib.Path(path)
    let parent = p.parent
    if parent != nil and str(parent) != "" { os.makedirs(str(parent)) }
    with open(path, "w") as f { f.write(text) }
}

func escape(s) {
    return s.replace("&", "&amp;").replace("<", "&lt;")
         .replace(">", "&gt;").replace('"', "&quot;")
}

func bold(s) {
    # convert **text** to <strong>text</strong>
    let out = ""
    var i = 0
    while i < len(s) {
        if i + 1 < len(s) and s[i] == '*' and s[i+1] == '*' {
            let end = s.find("**", i + 2)
            if end == -1 {
                out = out + s[i:]
                break
            }
            out = out + "<strong>" + escape(s[i+2:end]) + "</strong>"
            i = end + 2
        } else {
            out = out + escape(s[i])
            i += 1
        }
    }
    return out
}

func build_resume(text) {
    let lines = text.splitlines()
    let out = []
    out.push("""<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="utf-8">
<title>Resume</title>
<style>
@media print { body { margin: 0; } .no-print { display: none; } }
body { font-family: Georgia, serif; line-height: 1.5; color: #222;
       max-width: 800px; margin: 0 auto; padding: 2rem; }
h1 { font-size: 2.5rem; margin: 0 0 0.5rem 0; border-bottom: 2px solid #333;
     padding-bottom: 0.5rem; }
h2 { font-size: 1.4rem; margin: 1.5rem 0 0.5rem 0; color: #444;
     border-bottom: 1px solid #ccc; padding-bottom: 0.25rem; }
p, li { margin: 0.4rem 0; }
ul { margin: 0.4rem 0 0.8rem 1.2rem; }
.no-print { margin-top: 1rem; font-size: 0.9rem; color: #666; }
</style>
</head>
<body>
""")

    var in_section = false
    for line in lines {
        let s = line.strip()
        if s == "" { continue }
        if s.startsWith("# ") {
            out.push("<h1>" + bold(s[2:]) + "</h1>\n")
            in_section = false
        } elif s.startsWith("## ") {
            out.push("<h2>" + bold(s[3:]) + "</h2>\n")
            in_section = true
        } elif s.startsWith("- ") {
            if !in_section {
                out.push("<ul>\n")
                in_section = true
            }
            out.push("  <li>" + bold(s[2:]) + "</li>\n")
        } else {
            if in_section { out.push("</ul>\n"); in_section = false }
            out.push("<p>" + bold(s) + "</p>\n")
        }
    }
    if in_section { out.push("</ul>\n") }

    out.push("""
<div class="no-print">
<p>Generated by resume.z — print with Ctrl-P (background graphics enabled).</p>
</div>
</body>
</html>""")

    return out.join("")
}

func main(argv) {
    if len(argv) < 2 {
        print("Usage: zen run resume.z <input.md> [output.html]")
        print("  Example: zen run resume.z resume.md resume.html")
        return 1
    }
    let input = argv[1]
    let output = argv[2] ?? "resume.html"
    let text = read_all(input)
    let html = build_resume(text)
    write_all(output, html)
    print("wrote " + output)
    return 0
}

if __name__ == "__main__" {
    sys.exit(main(sys.argv))
}
```

```
# Ada Lovelace

Software Engineer | ada@example.com | (555) 123-4567
```

```
$ zen run resume.z resume.md resume.html
wrote resume.html
```

Open `resume.html` in a browser and print to PDF (Ctrl-P). The `@media print`
CSS removes margins and hides the "no-print" hint.

## 97.3 Design notes

- **Structure:** `# Name` → `<h1>`, `## Section` → `<h2>`, `- item` → `<li>`,
  everything else → `<p>`.
- **Bold:** `**text**` → `<strong>text</strong>`.
- **Print-friendly:** `@media print` sets zero page margins and hides the
  generator hint, so the printed resume is clean.
- **Single file:** no external CSS/JS dependencies.
